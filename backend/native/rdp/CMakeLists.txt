cmake_minimum_required(VERSION 3.24)
project(webnav_rdp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
# FreeRDP 3.x's Windows public header uses dllimport unless this is set,
# including for its supported static-library build.
add_compile_definitions(FREERDP_EXPORTS)
find_package(FreeRDP 3.32 REQUIRED)
find_package(FreeRDP-Client 3.32 REQUIRED)
find_package(WinPR 3.32 REQUIRED)
add_executable(webnav-rdp src/main.cpp)
target_compile_definitions(webnav-rdp PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(webnav-rdp PRIVATE /Zi)
  target_link_options(webnav-rdp PRIVATE /DEBUG /INCREMENTAL:NO /OPT:REF /OPT:ICF)
endif()
target_link_libraries(webnav-rdp PRIVATE freerdp-client freerdp winpr windowscodecs ole32 user32 crypt32)
if(WEBNAV_BUILD_TEST_PEER)
  add_executable(webnav-rdp-test-peer tests/peer.cpp)
  target_compile_definitions(webnav-rdp-test-peer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(webnav-rdp-test-peer PRIVATE freerdp winpr ws2_32)
endif()
